# J.A.R.V.I.S. AI Assistant — Python dependencies
# Install with:  pip install -r requirements.txt
#
# Tested on Windows with Python 3.13.
# On Python 3.13+ PyAudio needs the `audioop-lts` backport (included below).

requests
pyttsx3
SpeechRecognition
python-decouple
pywhatkit
PyAudio
audioop-lts
PyQt6
psutil
ddgs
Pillow

# Fully-local speech-to-text (optional): when installed, J.A.R.V.I.S. stops
# using Google's speech API and transcribes your voice offline. The model
# (~75 MB for "base") downloads on first use — configure via WHISPER_MODEL
# in .env (tiny/base/small/medium). Without it, Google's free API is used.
faster-whisper

# Exact volume control ("volume to 50%") — optional; volume up/down and
# mute work without it (Windows media keys).
pycaw

# Phone link — scan a QR (PHONE LINK button in the HUD) to use J.A.R.V.I.S.
# from your phone on the same Wi-Fi.
flask
qrcode[pil]
# cryptography generates the self-signed HTTPS cert that lets the phone
# browser use its microphone for voice input.
cryptography

# Offline wake-word detection ('hey jarvis'). numpy/onnxruntime are listed
# explicitly because the wake-word engine imports them directly. The pretrained
# model (~1.2 MB) is downloaded automatically on first run (see core/wake_word.py).
# If you skip openwakeword, the wake word falls back to online speech recognition.
openwakeword
numpy
onnxruntime
